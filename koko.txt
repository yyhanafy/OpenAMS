def validate_dc_propagation(dc_propagation: dict[str, Any]) -> list[str]:
    """Validate designer-authored ordered DC-propagation metadata."""
    errors: list[str] = []

    if not dc_propagation:
        return errors

    if dc_propagation.get("schema_version") != 1:
        errors.append("dc_propagation.schema_version must equal 1")

    if dc_propagation.get("execution") != "ordered":
        errors.append("dc_propagation.execution must equal 'ordered'")

    operations = dc_propagation.get("operations")
    if not isinstance(operations, list):
        errors.append("dc_propagation.operations must be a list")
        return errors

    allowed_types = {
        "equation",
        "technology_lookup",
        "lower_bound",
        "upper_bound",
    }

    seen_ids: set[str] = set()

    for index, operation in enumerate(operations):
        location = f"dc_propagation.operations[{index}]"

        if not isinstance(operation, dict):
            errors.append(f"{location} must be a mapping")
            continue

        operation_id = operation.get("id")
        operation_type = operation.get("type")
        target = operation.get("target")

        if not isinstance(operation_id, str) or not operation_id.strip():
            errors.append(f"{location}.id must be a nonempty string")
        elif operation_id in seen_ids:
            errors.append(
                f"{location}.id duplicates operation '{operation_id}'"
            )
        else:
            seen_ids.add(operation_id)

        if operation_type not in allowed_types:
            errors.append(
                f"{location}.type must be one of "
                f"{sorted(allowed_types)}"
            )

        if not isinstance(target, str) or not target.strip():
            errors.append(f"{location}.target must be a nonempty string")

        if operation_type in {
            "equation",
            "lower_bound",
            "upper_bound",
        }:
            expression = operation.get("expression")

                errors.append(

                )

    return errors

def _evaluate_dc_propagation_diagnostics(
    model: Mapping[str, Any],
    values: Mapping[str, float],
    provenance: Mapping[str, Any],
) -> dict[str, Any]:
    """Evaluate designer-authored DC propagation operations diagnostically.

    These results do not affect assignment acceptance yet.
    """

    propagation = (
        model.get("synthesis_interface", {})
        .get("dc_propagation", {})
        or {}
    )
    operations = propagation.get("operations", []) or []

    scope: dict[str, float] = {
        str(key): float(value)
        for key, value in values.items()
        if isinstance(value, (int, float))
    }

    operating = (
        model["project_inputs"]["design_rules"]
        .get("operating_conditions", {})
    )
    scope.update(
        {
            str(key): float(value)
            for key, value in operating.items()
            if isinstance(value, (int, float))
        }
    )

    for raw_device, item in provenance.items():
        if not isinstance(item, Mapping):
            continue

        vdsat = item.get("vdsat_v")
        if vdsat is None:
            continue

        device = str(raw_device).lower()
        value = float(vdsat)

        scope[f"vdsat_{device}"] = value
        scope[f"vdsat_{device}_v"] = value

    results: dict[str, Any] = {}

    for operation in operations:
        operation_id = str(operation.get("id", ""))
        operation_type = str(operation.get("type", ""))
        target = str(operation.get("target", ""))
        expression = operation.get("expression")

            record["error"] = str(exc)

            record["status"] = "evaluated"
            record["value"] = float(value)

    evaluated_values = [
        float(record["value"])

import json
import math
from pathlib import Path

generated = Path("examples/folded_cascode/generated")

topology = json.loads(
    (generated / "topology.json").read_text()
)

complete = json.loads(
    (
        generated
        / "assignment_synthesis"
        / "complete_assignments.json"
    ).read_text()
)

assignments = complete["assignments"]

# Collect actual topology-node names.
nodes = set()

for device in topology.get("devices", []):
    terminals = device.get("terminals", {})
    for node in terminals.values():
        if node:
            nodes.add(str(node).lower())

for port in topology.get("ports", []):
    if isinstance(port, str):
        nodes.add(port.lower())
    elif isinstance(port, dict) and port.get("name"):
        nodes.add(str(port["name"]).lower())

def assignment_keys_for_node(node):
    candidates = [
        f"{node}_v",
        node,
    ]

    # Existing Step-5 naming may drop a leading "v".
    if node.startswith("v") and len(node) > 1:
        candidates.append(f"{node[1:]}_v")

    return candidates

print("===== CIRCUIT NODE VOLTAGE RANGES =====")
print(f"assignments examined: {len(assignments)}")
print()

found_nodes = 0

for node in sorted(nodes):
    values = []

    for assignment in assignments:
        for key in assignment_keys_for_node(node):
            value = assignment.get(key)
            if isinstance(value, (int, float)) and math.isfinite(value):
                values.append(float(value))
                break

    if not values:
        continue

    found_nodes += 1
    print(
        f"{node:12s} "
        f"min={min(values): .9f} V  "
        f"max={max(values): .9f} V  "
        f"span={max(values) - min(values): .9f} V  "
        f"count={len(values)}"
    )

print()
print(f"topology nodes with resolved voltages: {found_nodes}")

# Report the new explicit propagation result for Vout separately.
lower_values = []
upper_values = []

for assignment in assignments:
    diagnostics = assignment.get(
        "dc_propagation_diagnostics", {}
    ).get("operations", {})

    lower = diagnostics.get(
        "output_headroom_lower", {}
    ).get("value")

    upper = diagnostics.get(
        "output_headroom_upper", {}
    ).get("value")

    if isinstance(lower, (int, float)):
        lower_values.append(float(lower))

    if isinstance(upper, (int, float)):
        upper_values.append(float(upper))

print()
print("===== PROPAGATED VOUT LIMITS =====")

if lower_values:
    print(
        "lower bound across assignments: "
        f"{min(lower_values):.9f} to "
        f"{max(lower_values):.9f} V"
    )
else:
    print("lower bound: unresolved")

if upper_values:
    print(
        "upper bound across assignments: "
        f"{min(upper_values):.9f} to "
        f"{max(upper_values):.9f} V"
    )
else:
    print("upper bound: unresolved")

valid_windows = sum(
    lower < upper
    for lower, upper in zip(lower_values, upper_values)
)

print(
    "nonempty propagated Vout windows: "
    f"{valid_windows}/{min(len(lower_values), len(upper_values))}"
)

import pandas as pd
from pathlib import Path

cases = [
    (
        "TWO_STAGE",
        Path("../MVP_archive_AUG_10/runtime/two_stage_all_2025_mlp_witnesses.csv"),
        Path("/tmp/two_stage_clean_100.csv"),
    ),
    (
        "FOLDED",
        Path("../MVP_archive_AUG_10/runtime/folded_cascode_all_2025_generic_mlp_witnesses.csv"),
        Path("/tmp/folded_clean_100.csv"),
    ),
]

def summarize(df):
    out = {}
    for p, g in df.groupby("point_index"):
        statuses = set(g["generation_status"].astype(str))
        has_witness = "WITNESS" in statuses

        best = None
        if has_witness:
            w = g[g["generation_status"] == "WITNESS"]
            if "max_abs_residual" in w.columns and len(w):
                best = pd.to_numeric(
                    w["max_abs_residual"],
                    errors="coerce"
                ).min()

        out[int(p)] = {
            "has_witness": has_witness,
            "best_residual": best,
        }
    return out

for name, old_path, new_path in cases:
    old = pd.read_csv(old_path)
    new = pd.read_csv(new_path)

    old_s = summarize(old)
    new_s = summarize(new)

    points = sorted(new_s)

    match = 0
    mismatch = []

    old_w = 0
    new_w = 0

    residual_pairs = []

    for p in points:
        o = old_s.get(p)
        n = new_s[p]

        if o is None:
            mismatch.append((p, "MISSING_OLD", n["has_witness"]))
            continue

        old_w += int(o["has_witness"])
        new_w += int(n["has_witness"])

        if o["has_witness"] == n["has_witness"]:
            match += 1
        else:
            mismatch.append(
                (p, o["has_witness"], n["has_witness"])
            )

        if (
            o["has_witness"]
            and n["has_witness"]
            and o["best_residual"] is not None
            and n["best_residual"] is not None
        ):
            residual_pairs.append(
                (p, o["best_residual"], n["best_residual"])
            )

    print()
    print("=" * 70)
    print(name)
    print("=" * 70)
    print("points compared:        ", len(points))
    print("classification matches: ", match)
    print("classification mismatch:", len(mismatch))
    print("old witness points:     ", old_w)
    print("new witness points:     ", new_w)

    if mismatch:
        print("\nMISMATCHES")
        for x in mismatch[:30]:
            print(" ", x)

    if residual_pairs:
        r = pd.DataFrame(
            residual_pairs,
            columns=["point_index", "old_best", "new_best"]
        )
        r["delta"] = r["new_best"] - r["old_best"]

        print("\nBEST-RESIDUAL COMPARISON")
        print("common witness points:", len(r))
        print("old mean best:        ", r["old_best"].mean())
        print("new mean best:        ", r["new_best"].mean())
        print("new better count:     ", (r["delta"] < 0).sum())
        print("new equal count:      ", (r["delta"].abs() < 1e-12).sum())
        print("new worse count:      ", (r["delta"] > 0).sum())

        print("\nLargest residual differences:")
        print(
            r.reindex(
                r["delta"].abs().sort_values(
                    ascending=False
                ).index
            ).head(10).to_string(index=False)
        )

 2921  python - <<'PY'
import pandas as pd
from pathlib import Path

base = Path("examples/two_stage_opamp/generated/assignment_synthesis/mlp_dataset_1500")

selected_csv = base / "selected_candidates.csv"
witness_csv  = Path("runtime/two_stage_all_2025_mlp_witnesses.csv")
out_csv      = base / "selected_witnesses_1500.csv"

sel = pd.read_csv(selected_csv)
wit = pd.read_csv(witness_csv)

selected_points = set(sel["point_index"].astype(int))

out = wit[wit["point_index"].astype(int).isin(selected_points)].copy()

# Keep the coverage-selection rank attached.
rank = sel[["point_index", "selection_rank"]].copy()
rank["point_index"] = rank["point_index"].astype(int)
out["point_index"] = out["point_index"].astype(int)

out = out.merge(rank, on="point_index", how="left")
out = out.sort_values(["selection_rank", "point_index"]).reset_index(drop=True)

out.to_csv(out_csv, index=False)

print("selected design points :", len(selected_points))
print("witness rows available :", len(out))
print("unique witness points  :", out["point_index"].nunique())
print("missing design points  :", len(selected_points) - out["point_index"].nunique())
print("output                 :", out_csv)

if "witness_index" in out.columns:
    print("\nwitnesses per selected point:")
    print(out.groupby("point_index").size().describe())
